⚡ Quick Start

# 1. Clone & install
git clone https://github.com/yourname/rag-llama3
cd rag-llama3
pip install -r requirements.txt

# 2. Configure
cp .env.example .env
# Edit .env — set MODEL_PATH, DATABASE_URL, AWS keys

# 3. Download model (GGUF INT4)
mkdir -p models
huggingface-cli download \
  bartowski/Meta-Llama-3-8B-Instruct-GGUF \
  Meta-Llama-3-8B-Instruct-Q4_K_M.gguf \
  --local-dir models/

# 4. Ingest documents
python scripts/ingest.py --source docs/ --chunk-size 512

# 5. Start API
uvicorn src.main:app --reload --port 8000

# Build and start all services
docker compose up --build

# In a second terminal — ingest docs
docker compose exec api \
  python scripts/ingest.py --source docs/

# Services:
#   API       → http://localhost:8000/docs
#   Prometheus → http://localhost:9090
#   Postgres  → localhost:5432

# Prerequisites: AWS CLI configured, SSH key exists
export KEY_NAME=rag-llama3-key
export KEY_PATH=~/.ssh/rag-llama3-key.pem

# One-command deploy to EC2 g4dn.xlarge
chmod +x scripts/deploy_ec2.sh
./scripts/deploy_ec2.sh

# Upload model to S3 (run once)
aws s3 cp models/llama-3-8b-instruct-q4_k_m.gguf \
  s3://$S3_BUCKET/models/

# Blocking query
curl -X POST http://localhost:8000/api/v1/query \
  -H "Content-Type: application/json" \
  -d '{"query": "What is INT4 quantisation?", "top_k": 5}'

# Streaming (SSE)
curl -N "http://localhost:8000/api/v1/query/stream?query=explain+FAISS"

# Submit feedback
curl -X POST http://localhost:8000/api/v1/feedback \
  -H "Content-Type: application/json" \
  -d '{"query":"...","answer":"...","rating":4,"comment":"Good but verbose"}'

# Health check
curl http://localhost:8000/health
